Let
\[
V_h=\operatorname{span}\{\varphi_1,\ldots,\varphi_N\}
\subset H_0^1(\Omega)
\]
be the fixed conforming finite-element space, and write
\[
u_h(\bU)=\sum_{j=1}^N U_j\varphi_j.
\]
The torsion function $\tau_h\in V_h$ is fixed, and the target torsion band is
\[
B_{\tau,h}
=
\{x\in\Omega:a_1<\tau_h(x)<a_2\},
\qquad
A_{\tau,h}=|B_{\tau,h}|.
\]
The goal is to determine the potential thresholds
\[
\bc=(c_1,c_2)\in\cC,
\qquad c_1<c_2,
\]
such that the localized semilinear equilibrium produces an active band contained in, and as close in area as possible to, $B_{\tau,h}$.

\section{Logistic-window nonlinearity}

Define the logistic function
\[
\sigma(z)=\frac{1}{1+e^{-z}},
\]
and the logistic window
\[
W(s;c_1,c_2,\varepsilon)
=
\sigma\left(\frac{s-c_1}{\varepsilon}\right)
-
\sigma\left(\frac{s-c_2}{\varepsilon}\right).
\]
The semilinear density is
\[
f_{\varepsilon,\bc}(s)
=
\rho_0 W(s;c_1,c_2,\varepsilon).
\]
Introduce
\[
\sigma_1(s)
=
\sigma\left(\frac{s-c_1}{\varepsilon}\right),
\qquad
\sigma_2(s)
=
\sigma\left(\frac{s-c_2}{\varepsilon}\right),
\]
and
\[
q_i(s)=\sigma_i(s)\bigl(1-\sigma_i(s)\bigr),
\qquad i=1,2.
\]
Then
\[
W=\sigma_1-\sigma_2,
\]
and the relevant derivatives are
\[
\frac{\partial W}{\partial s}
=
\frac{1}{\varepsilon}(q_1-q_2),
\]
\[
\frac{\partial W}{\partial c_1}
=
-\frac{1}{\varepsilon}q_1,
\qquad
\frac{\partial W}{\partial c_2}
=
\frac{1}{\varepsilon}q_2.
\]
Consequently,
\[
f_{\varepsilon,\bc}'(s)
=
\frac{\rho_0}{\varepsilon}(q_1-q_2),
\]
\[
\frac{\partial f_{\varepsilon,\bc}}{\partial c_1}(s)
=
-\frac{\rho_0}{\varepsilon}q_1,
\qquad
\frac{\partial f_{\varepsilon,\bc}}{\partial c_2}(s)
=
\frac{\rho_0}{\varepsilon}q_2.
\]
These formulas make both the Newton matrix and the parameter sensitivities explicit.

\section{Inner semilinear equilibrium problem}

For a fixed parameter pair $\bc$, seek $u_h\in V_h$ such that
\[
\int_\Omega \nabla u_h\cdot\nabla v_h\,dx
-
\rho_0
\int_\Omega
W(u_h;c_1,c_2,\varepsilon)v_h\,dx
=
0
\qquad
\forall v_h\in V_h.
\]
In coefficient form, define
\[
\br(\bU,\bc)
=
\bK\bU-\bF(\bU,\bc),
\]
where
\[
K_{ij}
=
\int_\Omega
\nabla\varphi_j\cdot\nabla\varphi_i\,dx,
\]
and
\[
F_i(\bU,\bc)
=
\rho_0
\int_\Omega
W(u_h(\bU);c_1,c_2,\varepsilon)
\varphi_i\,dx.
\]
The discrete equilibrium condition is
\[
\boxed{\br(\bU,\bc)=0.}
\]
For fixed $\bc$, Newton's method uses the Jacobian
\[
\bJ_U
=
\frac{\partial\br}{\partial\bU},
\]
whose entries are
\[
[\bJ_U]_{ij}
=
K_{ij}
-
\frac{\rho_0}{\varepsilon}
\int_\Omega
\bigl(q_1(u_h)-q_2(u_h)\bigr)
\varphi_j\varphi_i\,dx.
\]
The Newton correction satisfies
\[
\bJ_U\,\delta\bU
=
-\br.
\]
The natural residual merit function is
\[
\Psi_h(\bU;\bc)
=
\frac12
\br(\bU,\bc)^T
\bK^{-1}
\br(\bU,\bc).
\]
This merit function can be used for damping or line search, but the inner target remains
\[
\br(\bU,\bc)=0.
\]
Merely reaching a stationary point of $\Psi_h$ is not sufficient because
\[
\nabla_{\bU}\Psi_h
=
\bJ_U^T\bK^{-1}\br
\]
can vanish with $\br\neq0$ when $\bJ_U$ is singular or poorly conditioned.

We therefore define the branch-selected equilibrium map
\[
\bU=\bU(\bc)
\]
through the damped Newton procedure initialized from the previously computed localized equilibrium.

\section{Soft and certified active bands}

The logistic window itself provides a natural differentiable representation of the active band:
\[
\alpha_h(x;\bU,\bc)
=
W\bigl(u_h(x);c_1,c_2,\varepsilon\bigr).
\]
Inside the plateau,
\[
\alpha_h\approx1,
\]
while outside the band,
\[
\alpha_h\approx0.
\]
This soft activity field is used to construct differentiable discrepancy functionals.

For final geometric certification, define the interior active region by removing the logistic transition layers:
\[
B_h^\kappa(\bU,\bc)
=
\left\{
x\in\Omega:
c_1+\kappa\varepsilon
\leq u_h(x)\leq
c_2-\kappa\varepsilon
\right\},
\]
where $\kappa>0$ specifies how far inside each logistic transition the plateau begins.
The parameter set must therefore satisfy
\[
c_2-c_1>2\kappa\varepsilon.
\]
The soft window is used for optimization, while $B_h^\kappa$ is used to verify actual containment and area after convergence.

\section{Differentiable geometric discrepancy}

Let
\[
m_{\tau,h}(x)=\mathbf{1}_{B_{\tau,h}}(x)
\]
denote the fixed torsion-band indicator.

Define the soft leakage outside the torsion band by
\[
L_h(\bU,\bc)
=
\int_\Omega
\bigl(1-m_{\tau,h}(x)\bigr)
W(u_h;c_1,c_2,\varepsilon)\,dx,
\]
and the unfilled torsion-band area by
\[
M_h(\bU,\bc)
=
\int_\Omega
m_{\tau,h}(x)
\left[
1-W(u_h;c_1,c_2,\varepsilon)
\right]dx.
\]
The reduced functionals are
\[
\widehat L_h(\bc)
=
L_h(\bU(\bc),\bc),
\]
\[
\widehat M_h(\bc)
=
M_h(\bU(\bc),\bc).
\]
The outer optimization problem is
\[
\boxed{
\begin{aligned}
\min_{\bc\in\cC}\quad&
\widehat M_h(\bc),
\\
\text{subject to}\quad&
\widehat L_h(\bc)
\leq
\eta_{\mathrm{out}}A_{\tau,h},
\\
&
c_2-c_1
\geq
2\kappa\varepsilon+\delta_c.
\end{aligned}}
\]
Thus containment has priority, while the objective fills as much of the torsion band as possible.

\section{Derivatives of the geometric functionals}

Because
\[
\alpha_h=W(u_h;c_1,c_2,\varepsilon),
\]
its variation is
\[
\delta\alpha_h
=
W_s\,\delta u_h
+
W_{c_1}\,\delta c_1
+
W_{c_2}\,\delta c_2.
\]
Using the logistic formulas,
\[
\delta\alpha_h
=
\frac{q_1-q_2}{\varepsilon}\,\delta u_h
-
\frac{q_1}{\varepsilon}\,\delta c_1
+
\frac{q_2}{\varepsilon}\,\delta c_2.
\]
For the leakage functional,
\[
\delta L_h
=
\int_\Omega
(1-m_{\tau,h})
\delta\alpha_h\,dx.
\]
Hence
\[
\frac{\partial L_h}{\partial U_j}
=
\frac{1}{\varepsilon}
\int_\Omega
(1-m_{\tau,h})(q_1-q_2)\varphi_j\,dx,
\]
\[
\frac{\partial L_h}{\partial c_1}
=
-\frac{1}{\varepsilon}
\int_\Omega
(1-m_{\tau,h})q_1\,dx,
\]
and
\[
\frac{\partial L_h}{\partial c_2}
=
\frac{1}{\varepsilon}
\int_\Omega
(1-m_{\tau,h})q_2\,dx.
\]
Similarly,
\[
\frac{\partial M_h}{\partial U_j}
=
-\frac{1}{\varepsilon}
\int_\Omega
m_{\tau,h}(q_1-q_2)\varphi_j\,dx,
\]
\[
\frac{\partial M_h}{\partial c_1}
=
\frac{1}{\varepsilon}
\int_\Omega
m_{\tau,h}q_1\,dx,
\]
and
\[
\frac{\partial M_h}{\partial c_2}
=
-\frac{1}{\varepsilon}
\int_\Omega
m_{\tau,h}q_2\,dx.
\]
These derivatives are concentrated near the two logistic interfaces because $q_i$ is appreciable only when
\[
u_h\approx c_i.
\]
The outer search therefore receives information specifically from the two band boundaries, rather than from the entire domain uniformly.

\section{Equilibrium sensitivities with respect to $c_1,c_2$}

At a converged equilibrium,
\[
\br(\bU(\bc),\bc)=0.
\]
Differentiating with respect to $c_\alpha$ gives
\[
\bJ_U
\frac{\partial\bU}{\partial c_\alpha}
+
\frac{\partial\br}{\partial c_\alpha}
=
0.
\]
The residual derivatives are explicit. Since
\[
r_i
=
(\bK\bU)_i
-
\rho_0
\int_\Omega W(u_h;c_1,c_2,\varepsilon)\varphi_i\,dx,
\]
we obtain
\[
\frac{\partial r_i}{\partial c_1}
=
\frac{\rho_0}{\varepsilon}
\int_\Omega q_1(u_h)\varphi_i\,dx,
\]
and
\[
\frac{\partial r_i}{\partial c_2}
=
-\frac{\rho_0}{\varepsilon}
\int_\Omega q_2(u_h)\varphi_i\,dx.
\]
Define
\[
\bJ_c
=
\begin{bmatrix}
\partial_{c_1}\br&
\partial_{c_2}\br
\end{bmatrix}.
\]
The equilibrium sensitivity matrix is
\[
\boxed{
\bS
=
\frac{\partial\bU}{\partial\bc}
=
-\bJ_U^{-1}\bJ_c.
}
\]
Thus, after factorizing the Newton matrix, only two additional linear solves are required:
\[
\bJ_U\mathbf{s}_1
=
-\frac{\partial\br}{\partial c_1},
\]
\[
\bJ_U\mathbf{s}_2
=
-\frac{\partial\br}{\partial c_2}.
\]

\section{Reduced gradients in parameter space}

For any geometric functional $D_h(\bU,\bc)$, define
\[
\widehat D_h(\bc)
=
D_h(\bU(\bc),\bc).
\]
Its reduced gradient is
\[
\boxed{
\nabla_{\bc}\widehat D_h
=
\nabla_{\bc}D_h
+
\bS^T\nabla_{\bU}D_h.
}
\]
In particular,
\[
\nabla_{\bc}\widehat L_h
=
\nabla_{\bc}L_h
+
\bS^T\nabla_{\bU}L_h,
\]
and
\[
\nabla_{\bc}\widehat M_h
=
\nabla_{\bc}M_h
+
\bS^T\nabla_{\bU}M_h.
\]
The direct terms $\nabla_{\bc}D_h$ describe how moving $c_1,c_2$ changes the logistic window for a fixed $u_h$. The sensitivity terms $\bS^T\nabla_{\bU}D_h$ account for the fact that the equilibrium itself changes when the thresholds change.

Ignoring the sensitivity term would generally give an incorrect direction because $u_h$ may react strongly to changes in $c_1,c_2$.

\section{Constrained direction in $\cC$}

At outer iteration $k$, suppose the current equilibrium is approximately converged. Compute a parameter direction by solving the two-dimensional trust-region problem
\[
\begin{aligned}
\min_{\delta\bc\in\R^2}\quad&
\nabla\widehat M_h(\bc_k)^T\delta\bc
+
\frac12
\delta\bc^T\mathbf{B}_k\delta\bc,
\\
\text{subject to}\quad&
\widehat L_h(\bc_k)
+
\nabla\widehat L_h(\bc_k)^T\delta\bc
\leq
\eta_{\mathrm{out}}A_{\tau,h},
\\
&
\|\mathbf{D}_k\delta\bc\|
\leq
\Delta_k,
\\
&
c_{2,k}+\delta c_2
-
c_{1,k}-\delta c_1
\geq
2\kappa\varepsilon+\delta_c.
\end{aligned}
\]
Here:
\begin{itemize}[leftmargin=2em]
\item $\mathbf{B}_k$ is a positive-definite reduced Hessian approximation;
\item $\Delta_k$ is the trust radius;
\item $\mathbf{D}_k$ scales the two threshold variables.
\end{itemize}
Because this problem has only two unknowns, it can be solved directly and robustly.

When the current candidate is infeasible, the direction should first prioritize leakage reduction. One may solve
\[
\min_{\delta\bc}
\quad
\nabla\widehat L_h^T\delta\bc
+
\frac12\lambda\|\delta\bc\|^2.
\]
Once leakage is below tolerance, the missing-area functional becomes the principal objective.

\section{Sensitivity predictor}

After computing $\delta\bc_k$, define
\[
\bc_{k+1}^{\mathrm{trial}}
=
\bc_k+\delta\bc_k.
\]
Instead of restarting Newton at the old equilibrium $\bU_k$, use the sensitivity predictor
\[
\boxed{
\bU_{k+1}^{\mathrm{pred}}
=
\bU_k+\bS_k\delta\bc_k.
}
\]
Indeed,
\[
\bJ_U\bS+\bJ_c=0,
\]
so if
\[
\br(\bU_k,\bc_k)=0,
\]
then
\[
\br\left(
\bU_k+\bS_k\delta\bc_k,
\bc_k+\delta\bc_k
\right)
=
O(\|\delta\bc_k\|^2).
\]
The predicted field is therefore already close to equilibrium at the new threshold pair.

Starting from this predictor, perform a few damped Newton corrections at fixed $\bc_{k+1}^{\mathrm{trial}}$.

\section{Acceptance of a parameter step}

For the corrected trial equilibrium, compute
\[
\Theta_{\mathrm{trial}}
=
\sqrt{
\br_{\mathrm{trial}}^T
\bK^{-1}
\br_{\mathrm{trial}}
},
\]
\[
L_{\mathrm{trial}},
\qquad
M_{\mathrm{trial}},
\]
and the certified region
\[
B_{h,\mathrm{trial}}^\kappa.
\]
A trial step is accepted when:
\begin{enumerate}[leftmargin=2em]
\item the Newton residual satisfies the current inner tolerance;
\item leakage does not exceed the admissible filter envelope;
\item the missing-area or leakage discrepancy decreases as predicted;
\item the localized active band has not collapsed or jumped to another component.
\end{enumerate}
A useful branch-preservation test is
\[
\frac{
\int_\Omega
W(u_h^{\mathrm{trial}};\bc_{\mathrm{trial}},\varepsilon)
W(u_h^k;\bc_k,\varepsilon)\,dx
}{
\int_\Omega
W(u_h^k;\bc_k,\varepsilon)\,dx
}
\geq
\eta_{\mathrm{overlap}}.
\]
One can also impose
\[
\int_\Omega
W(u_h^{\mathrm{trial}};\bc_{\mathrm{trial}},\varepsilon)\,dx
\geq
A_{\min}
\]
to reject collapse toward the zero or low-amplitude branch.

\section{Inexact nested iteration}

It is unnecessary to solve the inner Newton problem to full precision at every outer iteration.
Let
\[
\Theta_k
=
\|\br(\bU_k,\bc_k)\|_{\bK^{-1}}.
\]
A practical inner tolerance is
\[
\Theta_k
\leq
\eta_k,
\]
where $\eta_k$ decreases as the geometric discrepancy decreases. For example,
\[
\eta_k
=
\min
\left\{
\eta_{\max},
\gamma
\left[
\frac{L_h}{A_{\tau,h}}
+
\frac{M_h}{A_{\tau,h}}
\right]
\right\}.
\]
Far from the desired band, Newton is solved only approximately. Near the optimum, the equilibrium residual is reduced more accurately.

This produces the desired simultaneous behavior:
\[
\text{reduce the semilinear residual}
\quad\text{and}\quad
\text{improve }(c_1,c_2)
\]
without completely solving one problem before making progress on the other.

\section{DOLFINx implementation strategy}

The executable implementation is
\[
\texttt{scripts/dolfinx\_torsion\_initialized\_window\_reduced\_optimization.py}.
\]
It uses a fixed DOLFINx Lagrange space on either a supplied Gmsh mesh or the
same generated smooth-star mesh used by the closed-loop refit runner.  The
script is deliberately organized around the twelve algorithmic steps listed
below.  With verbosity level \texttt{-v 2}, each outer iteration prints an
\texttt{ALGO\_STEP} line for these steps, including timings for the expensive
Newton, assembly, sensitivity-solve, and predictor-corrector phases.

The implementation uses three layers of residual control.
First, inner Newton projections during the outer loop use an adaptive tolerance
\[
\eta_k
=
\max\left\{
\mathrm{tol}_R,
\min\left(
\eta_{\max},
\gamma_{\mathrm{inner}}
\frac{L_h/A_{\tau,h}+M_h/A_{\tau,h}}
{\eta_{\mathrm{out}}+\mathrm{tol}_A}
\right)
\right\},
\]
where $\mathrm{tol}_R$, $\eta_{\max}$, and $\gamma_{\mathrm{inner}}$ correspond
to \texttt{--tol-res}, \texttt{--inner-tol-max}, and
\texttt{--inner-tol-gamma}.  Unless \texttt{--inner-newton-tol} is supplied for testing, this adaptive
rule controls the inner solves.  Second, trial
states must pass the residual filter before their threshold step can be
accepted.  Third, after the outer iteration stops for any reason, the final
state is always projected again to \texttt{--final-newton-tol-res} if supplied,
otherwise to \texttt{--tol-res}.  The reported solution is therefore always a
certified semilinear equilibrium when the final Newton solve succeeds.

The initial thresholds are selected by a projected candidate test rather than
by an algebraic fit alone.  The script first constructs a small fixed set of
candidate windows:
\begin{itemize}[leftmargin=2em]
\item the density least-squares window on $\phi_T=-\Delta^{-1}\rho_{\mathrm{design}}$;
\item target-weighted quantile windows, currently the central $5$--$95$ and
$10$--$90$ percent windows of $\phi_T$;
\item a target-weighted median window whose sampled activity area is matched
to $A_{\tau,h}$.
\end{itemize}
Each candidate is then Newton-projected from $\phi_T$ at fixed thresholds and
scored using the projected leakage, missing area, active-set overlap, and
relative density error.  The selected candidate contributes both its thresholds
and its projected state to the first outer iteration.  This extra startup work
is intentional: it avoids choosing a threshold pair that fits
$W(\phi_T;c_1,c_2,\varepsilon)$ but collapses to an unrelated semilinear branch
after projection.

There are two successful final outcomes.  The strict outcome is
\[
\texttt{CONVERGED},
\]
which means final Newton convergence plus
\[
L_h/A_{\tau,h}\leq\eta_{\mathrm{out}},
\qquad
M_h/A_{\tau,h}\leq \mathrm{tol}_A.
\]
The practical sub-band outcome is
\[
\texttt{CONVERGED\_CERTIFIED\_SUBBAND},
\]
which means final Newton convergence plus
\[
\frac{|B_h^\kappa\setminus B_{\tau,h}|}{A_{\tau,h}}
\leq
\eta_{\mathrm{out}},
\qquad
\frac{|B_h^\kappa|}{A_{\tau,h}}
\geq
f_{\mathrm{cert}}.
\]
Here $f_{\mathrm{cert}}$ is the value of
\texttt{--min-certified-area-fraction}.
Thus a run can be successful even when it does not fill the entire torsion
band, provided it produces a nontrivial certified equilibrium sub-band
contained in the torsion-initialized band.

\section{Complete algorithm}

Given an initial localized field $\bU_0$ and initial thresholds $\bc_0$:
\begin{enumerate}[leftmargin=2.2em]
\item At fixed $\bc_k$, apply damped Newton until
\[
\|\br_k\|_{\bK^{-1}}\leq\eta_k.
\]

\item Evaluate
\[
W,\quad q_1,\quad q_2,
\quad L_h,\quad M_h.
\]

\item Assemble the Newton matrix
\[
\bJ_U.
\]

\item Assemble
\[
\frac{\partial\br}{\partial c_1},
\qquad
\frac{\partial\br}{\partial c_2}.
\]

\item Solve
\[
\bJ_U\bS=-\bJ_c.
\]

\item Assemble
\[
\nabla_{\bU}L_h,
\quad
\nabla_{\bc}L_h,
\quad
\nabla_{\bU}M_h,
\quad
\nabla_{\bc}M_h.
\]

\item Form the reduced gradients
\[
\nabla\widehat L_h,
\qquad
\nabla\widehat M_h.
\]

\item Solve the two-dimensional constrained trust-region problem for
\[
\delta\bc_k.
\]

\item Predict the new equilibrium:
\[
\bU_{k+1}^{\mathrm{pred}}
=
\bU_k+\bS_k\delta\bc_k.
\]

\item Correct the predictor by damped Newton at
\[
\bc_{k+1}^{\mathrm{trial}}
=
\bc_k+\delta\bc_k.
\]

\item Accept or reject the trial pair using the true residual, leakage, missing area, active-band overlap, and certified active region.

\item Update the trust radius and repeat.
\end{enumerate}
The method stops when
\[
\|\br(\bU_k,\bc_k)\|_{\bK^{-1}}
\leq
\mathrm{tol}_R,
\]
\[
\frac{L_h(\bU_k,\bc_k)}{A_{\tau,h}}
\leq
\eta_{\mathrm{out}},
\]
and
\[
\frac{M_h(\bU_k,\bc_k)}{A_{\tau,h}}
\leq
\mathrm{tol}_A.
\]
The final result is then verified using the certified band
\[
B_h^\kappa
=
\left\{
c_1+\kappa\varepsilon
\leq u_h
\leq
c_2-\kappa\varepsilon
\right\}.
\]
If the strict missing-area target is not attainable on the selected
semilinear branch, the certified band can still provide the desired practical
outcome.  In that case the run is classified as a certified sub-band success
when final Newton convergence holds and
\[
\frac{|B_h^\kappa\setminus B_{\tau,h}|}{A_{\tau,h}}
\leq
\eta_{\mathrm{out}},
\qquad
\frac{|B_h^\kappa|}{A_{\tau,h}}
\geq
f_{\mathrm{cert}}.
\]
This criterion records the case where the equilibrium band is safely contained
in the torsion-initialized band and is not too thin, even though it does not
fill the entire target band.

The essential reduced-space structure is
\[
\boxed{
\bc
\longmapsto
\bU(\bc)
\longmapsto
W(u_h(\bc);\bc,\varepsilon)
\longmapsto
\text{torsion-band discrepancy}.
}
\]
Because the logistic derivatives are explicit, the outer direction requires no finite differences and no random sampling of $\cC$. Each outer iteration needs one Newton factorization, two sensitivity solves, and a two-dimensional constrained parameter update.
